```python
IMPLEMENTATION: Cascaded Selective Evaluation

Current logic:
  for rep in range(N_REPLICATES):
      score = judge.score(neuron, transcript)

New logic:
  - Initial 3 replications
  - Compute variance (disagreement metric)
  - If variance > threshold: escalate to K more
  - Return median of all replications
"""

# Create new module: src/trait/judge/cascade_eval.py

CODE_CASCADE = """
import numpy as np
from scipy import stats

def cascaded_evaluate(
    judge_client,
    neuron_id,
    transcript,
    target_phi=0.70,
    initial_reps=3,
    max_reps=10
):
    '''
    Cascaded selective evaluation for judge replication.
    
    Args:
        judge_client: Gemini judge instance
        neuron_id: Neuron to score
        transcript: Session transcript
        target_phi: D-study reliability target (e.g., 0.70)
        initial_reps: Start with N reps
        max_reps: Ceiling on total reps
    
    Returns:
        {
            'score': median_score,
            'confidence_interval': (ci_lo, ci_hi),
            'n_reps': actual_reps_used,
            'disagreement': variance(reps)
        }
    '''
    
    # Initial replications
    scores = []
    for i in range(initial_reps):
        score = judge_client.score(neuron_id, transcript)
        scores.append(score)
    
    # Compute disagreement (variance or IQR)
    disagreement = np.var(scores)
    
    # Decide to escalate based on disagreement and target Phi
    # Higher disagreement -> more confidence margin needed -> escalate
    phi_estimated = 1 - (disagreement / max_disagreement_observed)
    
    if phi_estimated < target_phi and len(scores) < max_reps:
        # Escalate: how many more reps?
        reps_needed = int(np.ceil((target_phi - phi_estimated) / some_sensitivity_factor))
        reps_needed = min(reps_needed, max_reps - initial_reps)
        
        for i in range(reps_needed):
            score = judge_client.score(neuron_id, transcript)
            scores.append(score)
    
    # Aggregate
    final_score = np.median(scores)
    ci = np.percentile(scores, [2.5, 97.5])  # 95% CI
    
    return {
        'score': final_score,
        'confidence_interval': ci,
        'n_reps': len(scores),
        'disagreement': np.var(scores)
    }
```

### Task W0.5 — Item #6 (Tobit Censored-EC)

**File:** `src/trait/extractors/per_dimension/ec_tobit.py` (new)

```python
"""
TOBIT CENSORED OBSERVATION MODEL FOR EC

Problem: EC scores are left-censored lower bounds (off-screen verification not observed)

Solution: 
  - Latent variable: verification_propensity (unobserved)
  - Observed: ec_score (censored at 0)
  - Fit Tobit regression
  - Output: asymmetric CI that widens when censoring suspected
"""

CODE_TOBIT = """
import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def fit_tobit_ec(ec_observed, covariates, initial_params=None):
    '''
    Fit Tobit model to EC scores.
    
    Model: 
        y_latent ~ Normal(X @ beta, sigma^2)
        y_observed = max(0, y_latent)
    
    Likelihood:
        P(y=0) = P(y_latent ≤ 0) = Φ(-X @ beta / sigma)
        P(y>0) = φ((y - X @ beta) / sigma) / sigma
    '''
    
    X = np.column_stack([np.ones(len(ec_observed)), covariates])
    y = ec_observed
    
    n_features = X.shape[1]
    
    def tobit_loglik(params):
        beta = params[:-1]
        sigma = params[-1]
        
        mu = X @ beta
        
        # Censored observations (y=0)
        censored = y == 0
        ll_censored = norm.logcdf(-mu[censored] / sigma).sum()
        
        # Uncensored observations (y>0)
        uncensored = y > 0
        ll_uncensored = norm.logpdf((y[uncensored] - mu[uncensored]) / sigma).sum()
        ll_uncensored -= uncensored.sum() * np.log(sigma)
        
        return -(ll_censored + ll_uncensored)
    
    # Fit
    if initial_params is None:
        initial_params = np.concatenate([np.zeros(n_features), [1.0]])
    
    result = minimize(tobit_loglik, initial_params, method='BFGS')
    
    # Extract parameters
    beta = result.x[:-1]
    sigma = result.x[-1]
    
    # Compute standard errors (Hessian)
    hess = result.hess_inv.todense() if hasattr(result.hess_inv, 'todense') else result.hess_inv
    se = np.sqrt(np.diag(hess))
    
    # Predicted latent values (for each observation)
    mu_pred = X @ beta
    
    # Credible intervals (95%)
    ci_lo = mu_pred - 1.96 * sigma
    ci_hi = mu_pred + 1.96 * sigma
    
    return {
        'beta': beta,
        'sigma': sigma,
        'std_errors': se,
        'mu_pred': mu_pred,
        'ci_lo': ci_lo,
        'ci_hi': ci_hi,
        'convergence': result.success
    }

def apply_tobit_to_ec_scores(gold_chats_26):
    '''
    Backfill existing 26-chat EC scores with Tobit censoring model.
    '''
    # Construct input
    ec_scores = []
    covariates = []  # e.g., session_length, archetype_coded, ...
    
    for chat in gold_chats_26:
        ec_score = chat.neurons['EC-5'].score  # Example
        ec_scores.append(ec_score)
        covariates.append([chat.session_length, chat.archetype_code, chat.language_code])
    
    # Fit model
    result = fit_tobit_ec(np.array(ec_scores), np.array(covariates))
    
    # Assign to scores table
    for i, chat in enumerate(gold_chats_26):
        chat.neurons['EC-5'].ci_lo = result['ci_lo'][i]
        chat.neurons['EC-5'].ci_hi = result['ci_hi'][i]
        chat.neurons['EC-5'].censored = True
        chat.neurons['EC-5'].point_estimate = ec_scores[i]
    
    return result
"""
```

```python
"""
IRT DIAGNOSTICS ON JUDGE OUTPUT

Purpose: Decompose judge error into noise vs. bias

Method: Fit Graded Response Model (GRM) to judge replications

CODE_JUDGE_IRT = """
from sklearn.linear_model import LogisticRegression
import numpy as np

def fit_grm_to_judge_reps(judge_replications, human_gold):
    '''
    Fit GRM to judge replications.
    
    Inputs:
        judge_replications: dict[neuron_id] -> array[scores from K reps]
        human_gold: dict[neuron_id] -> consensus human label (ground truth)
    
    Outputs:
        dict[neuron_id] -> {discrimination (a), difficulty (d), interpretation}
    '''
    
    results = {}
    
    for neuron_id, rep_scores in judge_replications.items():
        human_label = human_gold[neuron_id]
        
        # Aggregate judge consensus
        judge_consensus = np.median(rep_scores)
        
        # Ordinal logistic regression (proxy for GRM)
        X = judge_consensus.reshape(-1, 1)
        y = human_label
        
        # Fit logistic model
        clf = LogisticRegression(random_state=42)
        clf.fit(X, y)
        
        # Extract parameters
        a_param = clf.coef_[0][0]  # Discrimination (slope)
        d_param = clf.intercept_[0] / a_param  # Difficulty (threshold)
        
        # Diagnostic thresholds
        interpretation = ""
        if a_param < 0.8:
            interpretation += "LOW DISCRIMINATION: Rubric ambiguous, revise anchors. "
        if abs(d_param) > 2.0:
            interpretation += "DIFFICULTY EXTREME: Prompt too strict or too loose. "
        if interpretation == "":
            interpretation = "Acceptable. Judge functioning well."
        
        results[neuron_id] = {
            'discrimination': a_param,
            'difficulty': d_param,
            'judge_consensus': judge_consensus,
            'human_gold': human_label,
            'interpretation': interpretation
        }
    
    return results
```

```python
"""
κ^H EFFICIENCY TRAIT BUILD

κ^H = efficiency factor from IRT decomposition of S_human (tokens saved by human)

Inputs: 
  - S_human: information bottleneck measure
  - Covariates: session covariates (for falsification test)

Output:
  - κ^H trait estimate per session
  - Var(κ^H | model fixed): falsification gate
"""

CODE_KAPPA_H = """
import numpy as np
from scipy.optimize import minimize

def estimate_kappa_h(s_human_values, covariates, verbose=False):
    '''
    Estimate κ^H via information-theoretic efficiency decomposition.
    
    Model: 
      S_human = κ^H × λ_true + noise
    
    where κ^H is the efficiency factor (latent) and λ_true is learning rate.
    
    IRT approach: Fit logistic model to S_human as pseudo-item responses.
    '''
    
    # Standardize inputs
    X = (s_human_values - np.mean(s_human_values)) / np.std(s_human_values)
    
    # Fit 2PL IRT-like model
    def likelihood(params):
        a, b = params  # discrimination, difficulty
        predicted = 1.0 / (1.0 + np.exp(-a * (X - b)))
        
        # Log likelihood (Bernoulli)
        ll = np.sum(X * np.log(predicted + 1e-6) + (1 - X) * np.log(1 - predicted + 1e-6))
        return -ll
    
    result = minimize(likelihood, [1.0, 0.0], method='BFGS')
    
    a_param, b_param = result.x
    
    # κ^H is the extracted discrimination parameter
    kappa_h = a_param
    
    # Compute Var(κ^H | model fixed)
    predicted = 1.0 / (1.0 + np.exp(-a_param * (X - b_param)))
    residuals = X - predicted
    var_kappa = np.var(residuals)
    
    if verbose:
        print(f"κ^H estimate: {kappa_h:.3f}")
        print(f"Var(κ^H|model): {var_kappa:.3f}")
        print(f"Falsification gate: {'PASS' if var_kappa > 0.01 else 'FAIL (human signal is noise)'}")
    
    return {
        'kappa_h': kappa_h,
        'var_kappa': var_kappa,
        'falsification_pass': var_kappa > 0.01
    }

# Usage
corpus = load_gold_set(n=200)
s_human_all = [chat.s_human for chat in corpus]
covariates_all = [chat.covariates for chat in corpus]

result = estimate_kappa_h(np.array(s_human_all), covariates_all, verbose=True)

if not result['falsification_pass']:
    print("ERROR: Human signal collapsed. Framework falsified.")
    sys.exit(1)
```

```python
        "caveats": [

        ]
    }
    
    return report
```
